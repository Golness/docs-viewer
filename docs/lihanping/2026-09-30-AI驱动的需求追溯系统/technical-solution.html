1.2 目录解析实现

// 服务端：解析需求目录
function parseRequirementDir(dirName) {
  // 匹配格式：YYYY-MM-DD-标题
  const match = dirName.match(/^(\d{4}-\d{2}-\d{2})-(.+)$/);
  if (!match) return null;
  
  const [, date, title] = match;
  return {
    date,           // "2026-09-30"
    title,          // "优化慢任务"
    id: dirName     // "2026-09-30-优化慢任务"
  };
}

// 服务端：扫描所有需求
async function scanRequirements(docsDir) {
  const developers = await fs.readdir(docsDir);
  const requirements = [];
  
  for (const dev of developers) {
    const devPath = path.join(docsDir, dev);
    const stat = await fs.stat(devPath);
    if (!stat.isDirectory() || dev.startsWith('.')) continue;
    
    const dirs = await fs.readdir(devPath);
    for (const dir of dirs) {
      const parsed = parseRequirementDir(dir);
      if (!parsed) continue;
      
      const reqPath = path.join(devPath, dir);
      const files = await fs.readdir(reqPath);
      
      requirements.push({
        ...parsed,
        developer: dev,
        path: reqPath,
        files: files.filter(f => f.endsWith('.md')),
        hasMeta: files.includes('.meta.json')
      });
    }
  }
  
  // 按日期倒序排序
  return requirements.sort((a, b) => b.date.localeCompare(a.date));
}

2.1 目录树 API

// GET /docs-viewer/tree
async function handleTree(req, res) {
  const config = await loadConfig();
  const docsDir = path.resolve(config.docsDir);
  
  const tree = await buildTree(docsDir);
  json(res, { tree });
}

async function buildTree(rootPath) {
  const tree = { developers: [] };
  const developers = await fs.readdir(rootPath);
  
  for (const dev of developers) {
    if (dev.startsWith('.')) continue;
    
    const devPath = path.join(rootPath, dev);
    const stat = await fs.stat(devPath);
    if (!stat.isDirectory()) continue;
    
    const devNode = {
      name: dev,
      requirementCount: 0,    // 需求数量徽标
      requirements: []
    };
    
    const dirs = await fs.readdir(devPath);
    for (const dir of dirs) {
      const parsed = parseRequirementDir(dir);
      if (!parsed) continue;
      
      const reqPath = path.join(devPath, dir);
      const files = await fs.readdir(reqPath);
      
      const mdFiles = files.filter(f => f.endsWith('.md') || f.endsWith('.html'));
      const hasMeta = files.includes('.meta.json');
      
      // 获取 .meta.json 的修改时间（分析状态时间戳）
      let metaUpdatedAt = null;
      if (hasMeta) {
        const metaStat = await fs.stat(path.join(reqPath, '.meta.json'));
        metaUpdatedAt = metaStat.mtime.toISOString();
      }
      
      devNode.requirements.push({
        ...parsed,
        files: mdFiles.map(f => ({
          name: f,
          type: inferDocType(f)
        })),
        fileCount: mdFiles.length,    // 文档数量统计
        hasMeta,                       // .meta.json 状态标记
        metaUpdatedAt                  // 分析时间戳
      });
    }
    
    devNode.requirementCount = devNode.requirements.length;
    tree.developers.push(devNode);
  }
  
  return tree;
}

function inferDocType(filename) {
  const lower = filename.toLowerCase();
  if (lower.includes('requirement')) return 'requirement';
  if (lower.includes('design')) return 'design';
  if (lower.includes('cr')) return 'cr';
  return 'other';
}

2.2 文档读取 API

// GET /docs-viewer/read?path=xxx
async function handleRead(req, res) {
  const url = new URL(req.url, 'http://localhost');
  const filePath = url.searchParams.get('path');
  
  if (!filePath) {
    json(res, { error: 'path required' }, 400);
    return;
  }
  
  const config = await loadConfig();
  const docsDir = path.resolve(config.docsDir);
  const target = path.resolve(docsDir, filePath);
  
  // 安全检查：确保在 docsDir 内
  if (!target.startsWith(docsDir)) {
    json(res, { error: 'access denied' }, 403);
    return;
  }
  
  const content = await fs.readFile(target, 'utf8');
  json(res, { content, path: filePath });
}

3.2 精确搜索 API

// GET /docs-viewer/search?q=xxx&type=feature|field|tag
async function handleSearch(req, res) {
  const url = new URL(req.url, 'http://localhost');
  const query = url.searchParams.get('q');
  const type = url.searchParams.get('type') || 'all';
  
  if (!query) {
    json(res, { error: 'query required' }, 400);
    return;
  }
  
  const config = await loadConfig();
  const indexPath = path.join(config.docsDir, '.index', 'search-index.json');
  
  if (!fs.existsSync(indexPath)) {
    json(res, { error: 'index not built' }, 404);
    return;
  }
  
  const index = JSON.parse(await fs.readFile(indexPath, 'utf8'));
  const results = [];
  
  // 根据类型搜索
  if (type === 'all' || type === 'feature') {
    for (const [feature, reqIds] of Object.entries(index.featureIndex)) {
      if (feature.toLowerCase().includes(query.toLowerCase())) {
        results.push(...reqIds);
      }
    }
  }
  
  if (type === 'all' || type === 'field') {
    for (const [field, reqIds] of Object.entries(index.fieldIndex)) {
      if (field.toLowerCase().includes(query.toLowerCase())) {
        results.push(...reqIds);
      }
    }
  }
  
  if (type === 'all' || type === 'tag') {
    for (const [tag, reqIds] of Object.entries(index.tagIndex)) {
      if (tag.toLowerCase().includes(query.toLowerCase())) {
        results.push(...reqIds);
      }
    }
  }
  
  // 去重并获取完整需求信息
  const uniqueIds = [...new Set(results)];
  const matchedReqs = index.requirements.filter(r => uniqueIds.includes(r.id));
  
  // 附加源码引用片段（从索引中存储的 snippets 获取）
  const enrichResults = matchedReqs.map(req => ({
    ...req,
    snippets: index.snippets?.[req.id]?.filter(s =>
      s.text.toLowerCase().includes(query.toLowerCase())
    ).slice(0, 3) || []   // 最多返回 3 条引用
  }));
  
  json(res, { results: enrichResults, total: enrichResults.length });
}

3.3 AI 智能搜索 API

// POST /docs-viewer/ai-search
async function handleAiSearch(req, res) {
  const body = await readBody(req);
  const { query } = body;
  
  if (!query) {
    json(res, { error: 'query required' }, 400);
    return;
  }
  
  const config = await loadConfig();
  const indexPath = path.join(config.docsDir, '.index', 'search-index.json');
  const index = JSON.parse(await fs.readFile(indexPath, 'utf8'));
  
  // 构建索引摘要（只传关键信息，控制 token 数量）
  const indexSummary = index.requireations.map(r => ({
    id: r.id,
    title: r.title,
    summary: r.summary,
    features: r.features.slice(0, 5),  // 限制数量
    fields: r.fields.slice(0, 5),
    tags: r.tags
  }));
  
  const prompt = `你是一个需求追溯助手。用户输入查询词，你需要从索引中找到语义相关的需求。

查询词：${query}

索引摘要：
${JSON.stringify(indexSummary, null, 2)}

请返回 JSON 格式：
{
  "matches": [
    {
      "id": "需求ID",
      "score": 92,  // 0-100 匹配度
      "reason": "匹配原因"
    }
  ]
}

只返回与查询词相关的需求，不相关的不返回。`;
  
  try {
    const llm = ctx.llm;
    const stream = llm.stream({
      provider: config.ai?.provider || 'deepseek',
      model: config.ai?.model || 'deepseek-chat',
      messages: [{ role: 'user', content: prompt }],
      temperature: 0.3,
      maxTokens: 2000
    });
    
    let content = '';
    for await (const chunk of stream) {
      if (chunk.type === 'text-delta') {
        content += chunk.text;
      }
    }
    
    // 解析 AI 响应
    const aiResult = parseAiResponse(content);
    
    // 关联完整需求信息
    const results = aiResult.matches.map(m => {
      const req = index.requirements.find(r => r.id === m.id);
      return req ? { ...req, score: m.score, reason: m.reason } : null;
    }).filter(Boolean);
    
    json(res, { results, total: results.length });
  } catch (err) {
    json(res, { error: 'AI search failed: ' + err.message }, 500);
  }
}

function parseAiResponse(content) {
  try {
    // 尝试直接解析
    return JSON.parse(content);
  } catch {
    // 尝试提取 JSON 代码块
    const match = content.match(/```json\n([\s\S]+?)\n```/);
    if (match) {
      return JSON.parse(match[1]);
    }
    // 降级
    return { matches: [] };
  }
}

  json(res, { success: true, path: targetPath });
}

4.2 AI 分析需求 API

// POST /docs-viewer/analyze
async function handleAnalyze(req, res) {
  const body = await readBody(req);
  const { requirementId } = body;
  
  const config = await loadConfig();
  const docsDir = path.resolve(config.docsDir);
  
  // 查找需求目录
  const reqPath = findRequirementPath(docsDir, requirementId);
  if (!reqPath) {
    json(res, { error: 'requirement not found' }, 404);
    return;
  }
  
  // 读取所有 .md 文件
  const files = await fs.readdir(reqPath);
  const mdFiles = files.filter(f => f.endsWith('.md'));
  
  let docsContent = '';
  for (const file of mdFiles) {
    const content = await fs.readFile(path.join(reqPath, file), 'utf8');
    docsContent += `\n\n## ${file}\n${content}`;
  }
  
  // 截断过长内容（避免超出 context window）
  if (docsContent.length > 30000) {
    docsContent = docsContent.slice(0, 30000) + '\n\n[内容已截断]';
  }
  
  // 构建 Prompt
  const prompt = config.ai?.prompts?.analyze || DEFAULT_ANALYZE_PROMPT;
  const fullPrompt = `${prompt}\n\n${docsContent}`;
  
  try {
    const llm = ctx.llm;
    const stream = llm.stream({
      provider: config.ai?.provider || 'deepseek',
      model: config.ai?.model || 'deepseek-chat',
      messages: [{ role: 'user', content: fullPrompt }],
      temperature: 0.3,
      maxTokens: 4000
    });
    
    let content = '';
    for await (const chunk of stream) {
      if (chunk.type === 'text-delta') {
        content += chunk.text;
      }
    }
    
    // 解析并校验
    const meta = parseAndValidateMeta(content);
    
    // 原子写入 .meta.json
    const metaPath = path.join(reqPath, '.meta.json');
    const tmpPath = metaPath + '.tmp';
    await fs.writeFile(tmpPath, JSON.stringify(meta, null, 2));
    await fs.rename(tmpPath, metaPath);
    
    json(res, { success: true, meta });
  } catch (err) {
    json(res, { error: 'analyze failed: ' + err.message }, 500);
  }
}

    {
      "date": "YYYY-MM-DD",
      "type": "requirement|design|cr|release",
      "title": "事件标题",
      "description": "事件描述"
    }
  ]
}

只返回 JSON，不要其他内容。`;

function parseAndValidateMeta(content) {
  let meta;
  
  // 三级容错
  try {
    // 1. 直接解析
    meta = JSON.parse(content);
  } catch {
    // 2. 提取 JSON 代码块
    const match = content.match(/```json\n([\s\S]+?)\n```/);
    if (match) {
      meta = JSON.parse(match[1]);
    } else {
      // 3. 降级默认值
      meta = {
        type: 'unknown',
        summary: '',
        features: [],
        fields: [],
        tags: [],
        timeline: []
      };
    }
  }
  
  // Schema 校验（简化版）
  if (!meta.type) meta.type = 'unknown';
  if (!meta.summary) meta.summary = '';
  if (!Array.isArray(meta.features)) meta.features = [];
  if (!Array.isArray(meta.fields)) meta.fields = [];
  if (!Array.isArray(meta.tags)) meta.tags = [];
  if (!Array.isArray(meta.timeline)) meta.timeline = [];
  
  return meta;
}

// POST /docs-viewer/rebuild-index
async function handleRebuildIndex(req, res) {
  const config = await loadConfig();
  const docsDir = path.resolve(config.docsDir);
  
  // 扫描所有需求
  const requirements = await scanRequirements(docsDir);
  
  // 构建索引
  const index = {
    version: 1,
    builtAt: new Date().toISOString(),
    requirements: [],
    featureIndex: {},
    fieldIndex: {},
    tagIndex: {}
  };
  
  for (const req of requirements) {
    if (!req.hasMeta) continue;
    
    const metaPath = path.join(req.path, '.meta.json');
    const meta = JSON.parse(await fs.readFile(metaPath, 'utf8'));
    
    const reqEntry = {
      id: req.id,
      developer: req.developer,
      date: req.date,
      title: req.title,
      type: meta.type,
      summary: meta.summary,
      features: meta.features,
      fields: meta.fields,
      tags: meta.tags
    };
    
    index.requirements.push(reqEntry);
    
    // 构建倒排索引
    for (const feature of meta.features) {
      if (!index.featureIndex[feature]) {
        index.featureIndex[feature] = [];
      }
      index.featureIndex[feature].push(req.id);
    }
    
    for (const field of meta.fields) {
      if (!index.fieldIndex[field]) {
        index.fieldIndex[field] = [];
      }
      index.fieldIndex[field].push(req.id);
    }
    
    for (const tag of meta.tags) {
      if (!index.tagIndex[tag]) {
        index.tagIndex[tag] = [];
      }
      index.tagIndex[tag].push(req.id);
    }
  }
  
  // 写入索引文件
  const indexPath = path.join(docsDir, '.index', 'search-index.json');
  await fs.mkdir(path.dirname(indexPath), { recursive: true });
  await fs.writeFile(indexPath, JSON.stringify(index, null, 2));
  
  json(res, {
    success: true,
    stats: {
      requirements: index.requirements.length,
      features: Object.keys(index.featureIndex).length,
      fields: Object.keys(index.fieldIndex).length,
      tags: Object.keys(index.tagIndex).length
    }
  });
}

5.1 ctx.llm 调用模式

// 服务端：封装 LLM 调用
async function callLLM(prompt, config) {
  const llm = ctx.llm;
  
  const stream = llm.stream({
    provider: config.ai?.provider || 'deepseek',
    model: config.ai?.model || 'deepseek-chat',
    messages: [{ role: 'user', content: prompt }],
    temperature: 0.3,
    maxTokens: 4000
  });
  
  let content = '';
  let usage = null;
  let finishReason = null;
  
  for await (const chunk of stream) {
    if (chunk.type === 'text-delta') {
      content += chunk.text;
    } else if (chunk.type === 'usage') {
      usage = chunk.usage;
    } else if (chunk.type === 'finish') {
      finishReason = chunk.reason;
    }
  }
  
  // 检查异常结束
  if (finishReason?.kind === 'error') {
    throw new Error(`LLM error: ${finishReason.failure.message}`);
  }
  
  if (finishReason?.kind === 'max-tokens') {
    console.warn('LLM output truncated due to max-tokens limit');
  }
  
  return { content, usage, finishReason };
}

// 增量更新索引（分析单个需求后）
async function updateIndexEntry(docsDir, reqId, meta) {
  const index = await loadIndex(docsDir);
  
  if (!index) {
    // 索引不存在，跳过
    return;
  }
  
  // 删除旧条目
  index.requirements = index.requirements.filter(r => r.id !== reqId);
  
  // 清理倒排索引
  for (const idx of Object.keys(index.featureIndex)) {
    index.featureIndex[idx] = index.featureIndex[idx].filter(id => id !== reqId);
    if (index.featureIndex[idx].length === 0) {
      delete index.featureIndex[idx];
    }
  }
  // 同样处理 fieldIndex 和 tagIndex...
  
  // 添加新条目
  const req = findRequirementById(docsDir, reqId);
  index.requirements.push({
    id: reqId,
    developer: req.developer,
    date: req.date,
    title: req.title,
    type: meta.type,
    summary: meta.summary,
    features: meta.features,
    fields: meta.fields,
    tags: meta.tags
  });
  
  // 更新倒排索引
  for (const feature of meta.features) {
    if (!index.featureIndex[feature]) {
      index.featureIndex[feature] = [];
    }
    index.featureIndex[feature].push(reqId);
  }
  // 同样处理 fields 和 tags...
  
  // 原子写入
  const indexPath = getIndexPath(docsDir);
  const tmpPath = indexPath + '.tmp';
  await fs.writeFile(tmpPath, JSON.stringify(index, null, 2));
  await fs.rename(tmpPath, indexPath);
}

      features: Object.keys(index.featureIndex).length,
      fields: Object.keys(index.fieldIndex).length,
      tags: Object.keys(index.tagIndex).length
    }
  });
}

    try {

      content = aiContent;
    } catch (err) {
      // AI 失败时降级为静态模板
      addLog('warning', `⚠️ AI 生成失败，降级为静态模板: ${err.message}`);
